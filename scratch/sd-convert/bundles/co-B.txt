subgraph AP["Consensus Anti-Patterns and Consequences"]
  direction TB
  subgraph R1[" "]
    direction TB
    subgraph A1["Anti-Pattern 1: Consensus for Everything"]
      @L a1 Symptom: High latency for trivial reads, frequent compaction;;Fix: Use consensus only for strongly-consistent data
    end
    subgraph A2["Anti-Pattern 2: Single-Node Cluster"]
      @L a2 Symptom: No fault tolerance, single point of failure;;Fix: Use 3+ nodes or skip consensus entirely
    end
    subgraph A3["Anti-Pattern 3: Cross-Ocean Deployment"]
      @L a3 Symptom: 200-300ms write latency, timeouts;;Fix: Keep quorum in one region, use async replicas elsewhere
    end
  end
  subgraph R2[" "]
    direction TB
    subgraph A4["Anti-Pattern 4: Disabled Fsync"]
      @L a4 Symptom: Data loss after power failure, violated guarantees;;Fix: Enable fsync, use SSDs, batch fsyncs
    end
    subgraph A5["Anti-Pattern 5: No Compaction"]
      @L a5 Symptom: Disk full, node crash, cluster unavailable;;Fix: Auto-compaction with retention policy, disk monitoring
    end
  end
  R1 ~~~ R2
end
