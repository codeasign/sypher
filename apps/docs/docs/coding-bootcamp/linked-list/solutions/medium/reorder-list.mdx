# Reorder List — Solution

## Approach: Find Middle, Reverse, Merge

**Idea:** Three-step process: find the middle, reverse the second half, then interleave the two halves.

**Algorithm:**
1. Find middle using slow/fast pointers.
2. Reverse the second half.
3. Merge the two halves by alternating nodes.

**Complexity:** O(n) time, O(1) space.

def reorder_list(head: list[int]) -> list[int]:
    if not head: return []
    # Find middle
    mid = (len(head) + 1) // 2
    first = head[:mid]
    second = head[mid:][::-1]
    # Interleave
    result = []
    i, j = 0, 0
    while i < len(first) and j < len(second):
        result.append(first[i]); i += 1
        if j < len(second): result.append(second[j]); j += 1
    while i < len(first): result.append(first[i]); i += 1
    return result

class Solution {
    public void reorderList(ListNode head) {
        if (head == null) return;
        // Find middle
        ListNode slow = head, fast = head;
        while (fast.next != null && fast.next.next != null) {
            slow = slow.next;
            fast = fast.next.next;
        }
        // Reverse second half
        ListNode prev = null, curr = slow.next;
        while (curr != null) {
            ListNode next = curr.next;
            curr.next = prev;
            prev = curr;
            curr = next;
        }
        slow.next = null;
        // Merge two halves
        ListNode first = head, second = prev;
        while (second != null) {
            ListNode tmp1 = first.next, tmp2 = second.next;
            first.next = second;
            second.next = tmp1;
            first = tmp1;
            second = tmp2;
        }
    }
}

pub fn reorder_list(head: &mut Vec<i32>) {
    if head.is_empty() {
        return;
    }
    let mid = (head.len() + 1) / 2;
    let first: Vec<i32> = head[..mid].to_vec();
    let mut second: Vec<i32> = head[mid..].to_vec();
    second.reverse();

    let mut result = Vec::with_capacity(head.len());
    let mut i = 0;
    let mut j = 0;
    while i < first.len() && j < second.len() {
        result.push(first[i]);
        i += 1;
        if j < second.len() {
            result.push(second[j]);
            j += 1;
        }
    }
    while i < first.len() {
        result.push(first[i]);
        i += 1;
    }

    *head = result;
}

func reorderList(head *ListNode) {
    if head == nil {
        return
    }
    slow, fast := head, head
    for fast.Next != nil && fast.Next.Next != nil {
        slow = slow.Next
        fast = fast.Next.Next
    }
    var prev *ListNode
    curr := slow.Next
    for curr != nil {
        next := curr.Next
        curr.Next = prev
        prev = curr
        curr = next
    }
    slow.Next = nil
    first, second := head, prev
    for second != nil {
        tmp1, tmp2 := first.Next, second.Next
        first.Next = second
        second.Next = tmp1
        first = tmp1
        second = tmp2
    }
}
